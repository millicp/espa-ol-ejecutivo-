            <button class="btn" onclick="prevCard()">Anterior</button>
            <button class="btn" onclick="nextCard()">Siguiente</button>

        <div class="quiz-card">

            <div class="options-grid" id="q-options"></div>
        </div>
        <button class="btn" id="next-q-btn" style="display:none;" onclick="nextQuestion()">Siguiente Pregunta</button>
        <div class="score-box" id="score-display">Puntuación: 0 / 5</div>
    </div>

    <script>
        // Datos de Flashcards
        const cardsData = [
            { term: "firmar", collab: "firmar un contrato comercial", trans: "to sign a commercial contract" },
            { term: "enviar", collab: "enviar un correo electrónico", trans: "to send an email" },
            { term: "preparar", collab: "preparar la reunión de directorio", trans: "to prepare the board meeting" },
            { term: "visitar", collab: "visitar las instalaciones", trans: "to visit the facilities" },
            { term: "negociar", collab: "negociar los términos del acuerdo", trans: "to negotiate agreement terms" },
            { term: "leer", collab: "leer el reporte de auditoría", trans: "to read the audit report" },
            { term: "vender", collab: "vender servicios corporativos", trans: "to sell corporate services" },
            { term: "abrir", collab: "abrir una nueva filial", trans: "to open a new branch" }
        ];

        let currentCardIndex = 0;

        function flipCard() {
            document.getElementById('card').classList.toggle('is-flipped');
        }

        function updateCard() {
            const cardEl = document.getElementById('card');
            cardEl.classList.remove('is-flipped');
            setTimeout(() => {
                document.getElementById('fc-term').innerText = cardsData[currentCardIndex].term;
                document.getElementById('fc-collab').innerText = cardsData[currentCardIndex].collab;
                document.getElementById('fc-trans').innerText = cardsData[currentCardIndex].trans;
            }, 150);
        }

        function nextCard() {
            currentCardIndex = (currentCardIndex + 1) % cardsData.length;
            updateCard();
        }

        function prevCard() {
            currentCardIndex = (currentCardIndex - 1 + cardsData.length) % cardsData.length;
            updateCard();
        }

        // Datos del Quiz
        const quizData = [
            {
                q: "Para cerrar el acuerdo hoy, el CEO necesita ___________ el contrato comercial.",
                options: ["firmar", "enviar", "visitar"],
                correct: 0
            },
            {
                q: "La secretaria va a ___________ un correo electrónico al cliente VIP.",
                options: ["preparar", "enviar", "leer"],
                correct: 1
            },
            {
                q: "Antes de la presentación, el equipo debe ___________ la reunión de directorio.",
                options: ["negociar", "preparar", "abrir"],
                correct: 1
            },
            {
                q: "El analista necesita ___________ el informe de auditoría antes de la junta.",
                options: ["leer", "vender", "firmar"],
                correct: 0
            },
            {
                q: "La empresa planea ___________ una nueva filial en Lima este trimestre.",
                options: ["enviar", "visitar", "abrir"],
                correct: 2
            }
        ];

        let currentQuizIndex = 0;
        let score = 0;

        function loadQuiz() {
            if (currentQuizIndex >= quizData.length) {
                document.getElementById('q-text').innerText = "¡Reto completado con éxito!";
                document.getElementById('q-options').innerHTML = `<p style="text-align:center; font-size:1.1rem;">Tu puntuación final es ${score} de ${quizData.length}</p>`;
                document.getElementById('next-q-btn').style.display = 'none';
                return;
            }

            const currentQ = quizData[currentQuizIndex];
            document.getElementById('q-text').innerText = `${currentQuizIndex + 1}. ${currentQ.q}`;
            
            const optionsContainer = document.getElementById('q-options');
            optionsContainer.innerHTML = '';

            currentQ.options.forEach((opt, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.innerText = opt;
                btn.onclick = () => selectOption(index, btn, currentQ.correct);
                optionsContainer.appendChild(btn);
            });

            document.getElementById('next-q-btn').style.display = 'none';
        }

        function selectOption(selectedIndex, buttonElement, correctIndex) {
            const buttons = document.querySelectorAll('.option-btn');
            buttons.forEach(btn => btn.disabled = true);

            if (selectedIndex === correctIndex) {
                buttonElement.classList.add('correct');
                score++;
            } else {
                buttonElement.classList.add('incorrect');
                buttons[correctIndex].classList.add('correct');
            }

            document.getElementById('score-display').innerText = `Puntuación: ${score} / ${quizData.length}`;
            document.getElementById('next-q-btn').style.display = 'block';
        }

        function nextQuestion() {
            currentQuizIndex++;
            loadQuiz();
        }

        function switchTab(tabId, evt) {
            document.querySelectorAll('.container').forEach(c => c.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));

            document.getElementById(tabId).classList.add('active');
            evt.currentTarget.classList.add('active');

            if (tabId === 'quiz' && currentQuizIndex === 0) {
                loadQuiz();
            }
        }

        // Inicializar
        updateCard();
    </script>
